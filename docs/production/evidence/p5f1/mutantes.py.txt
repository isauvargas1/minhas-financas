def digest(path):
  return hashlib.sha256(path.read_bytes()).hexdigest()

def run(cmd):
  return subprocess.run(cmd, cwd=WT, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL).returncode

LOADERS = {'.ts': 'ts', '.tsx': 'tsx', '.mjs': 'js', '.js': 'js'}
CHECK = ("const e=require(process.argv[1]);let s='';process.stdin.on('data',d=>s+=d)"
         ".on('end',()=>{try{e.transformSync(s,{loader:process.argv[2]});process.exit(0)}catch(_){process.exit(1)}})")

def compiles(path, text):
  """Sintaxe do arquivo mutado (esbuild, o mesmo da build): mutante que não compila não conta como morto."""
  loader = LOADERS.get(path.suffix)
  if loader is None:
    return True  # firestore.rules: a guarda lê o texto; as Rules mutadas passam pelo carregamento no Emulator
  check = subprocess.run(['node', '-e', CHECK, str(WT / 'node_modules/esbuild'), loader], input=text.encode('utf-8'),
                         stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
  return check.returncode == 0

def apply_and_test(group, rel, old, new, cmd):
  path = WT / rel
  original = path.read_bytes()
  before = digest(path)
  text = original.decode('utf-8')
  if text.count(old) != 1:
    return group, rel, old, 'INVÁLIDO (trecho ausente ou repetido)'
  mutated = text.replace(old, new, 1)
  if not compiles(path, mutated):
    return group, rel, old, 'NÃO COMPILA'
  path.write_text(mutated, encoding='utf-8')
  try:
    code = run(cmd)
  finally:
    path.write_bytes(original)
  assert digest(path) == before, f'{rel} não voltou ao original'
  return group, rel, old, 'MORTO' if code != 0 else 'SOBREVIVEU'

# MUTANTES_SO=153,176: só os mutantes de unidade com esses números (1 = o primeiro da lista), para a rodada dos
# sobreviventes depois de testes novos; a saída marca cada um com o número.
ONLY = {int(item) for item in os.environ.get('MUTANTES_SO', '').split(',') if item.strip()}


def main():
  rows = []
  if MODE == 'unit':
    chosen = [(index, mutant) for index, mutant in enumerate(UNIT, 1) if not ONLY or index in ONLY]
    assert not ONLY or len(chosen) == len(ONLY), ('número de mutante inexistente', sorted(ONLY))
    baseline = run(NODE + sorted({f for _, m in chosen for f in m[4]}))
    if baseline != 0:
      print('linha de base falhou: os testes precisam passar antes dos mutantes'); sys.exit(2)
    for index, (group, rel, old, new, files) in chosen:
      row = apply_and_test(group, rel, old, new, NODE + files)
      rows.append((f'U{index} {row[0]}', *row[1:]) if ONLY else row)
  else:
    rules_cmd = ['node', '--test', 'tests/firestore/work-time-p5f.rules.integration.test.mjs']
    if run(rules_cmd) != 0:
      print('linha de base das Rules falhou'); sys.exit(2)
    for group, rel, old, new in RULES:
      rows.append(apply_and_test(group, rel, old, new, rules_cmd))
  dead = sum(1 for row in rows if row[3] == 'MORTO')
  broken = sum(1 for row in rows if row[3] == 'NÃO COMPILA')
  for group, rel, old, result in rows:
    first = old.strip().splitlines()[0][:90]
    print(f'{result:11} {group:11} {rel}: {first}')
  print(f'TOTAL {MODE}: {dead}/{len(rows)} mortos; {broken} não compilam')
  sys.exit(0 if dead == len(rows) else 1)

main()
