#!/usr/bin/env bash
# P5-F.1. Uso: run-pw-tree.sh <diretório da árvore> <rótulo> [args do playwright...]
# Igual ao run-pw.sh, para outra árvore (a da P5-F, na comparação antes × depois): build E2E
# (SKIP_BUILD=1 reaproveita o dist), Playwright contra o Emulator (minhas-financas-local) com o
# loopback-proxy seguro e, no fim, só a JVM órfã desta execução encerrada (identidade e cwd).
set -u
S=<scratchpad>
WT=$1; LABEL=$2; shift 2
L=$S/p5f1/logs; mkdir -p "$L"
START=$(date +%s)
export PLAYWRIGHT_BROWSERS_PATH=$S/pw-browsers
if [ "${SKIP_BUILD:-0}" != 1 ]; then cd "$WT" && npm run build:e2e > "$L/$LABEL-build.log" 2>&1; echo "build=$?"; fi
cd "$WT" && timeout 5400 bash $S/with-proxy-safe.sh 38831 npx playwright test "$@" > "$L/$LABEL.log" 2>&1
echo "exit=$?  duração=$(( $(date +%s) - START ))s"
sed -E 's/\x1b\[[0-9;]*[A-Za-z]//g' "$L/$LABEL.log" | grep -E "^\s+[0-9]+ (passed|failed|skipped|flaky|did not run)|✘|^\s+Expected|^\s+Received|Error: |at .*spec\.ts:[0-9]+" | head -30
bash $S/p5e11/stop-own-emulator.sh 8080 "$START" "$L/emulator-cleanup.log" "$WT" >/dev/null 2>&1; echo "limpeza=$? java=$(pgrep -c java || true)"
