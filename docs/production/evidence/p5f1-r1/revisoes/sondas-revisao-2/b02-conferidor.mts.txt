// Revisão 2, B02: solidez do conferidor do calendário inteiro (calendarAmbiguity, com o pré-filtro `covered`).
// O que ele aceita nunca pode ter dia ambíguo em nenhuma janela que a estimativa use: qualquer mês de 12 meses antes
// do mês de hoje em diante, como mês avulso e abrindo o período (o 1º mês usado pode ser qualquer um da janela).
// Uso: node ... b02-conferidor.mts <raiz> [amostras] [semente]
const root = process.argv[2];
const SAMPLES = Number(process.argv[3] ?? 800);
let seed = Number(process.argv[4] ?? 5);
const cal = await import(`${root}/src/modules/work-time/calendar.ts`);
const dates = await import(`${root}/src/modules/work-time/dates.ts`);
const rnd = () => { seed = (seed + 0x6d2b79f5) | 0; let v = Math.imul(seed ^ (seed >>> 15), 1 | seed); v = (v + Math.imul(v ^ (v >>> 7), 61 | v)) ^ v; return ((v ^ (v >>> 14)) >>> 0) / 4294967296; };
const int = (a: number, b: number) => a + Math.floor(rnd() * (b - a + 1));
const pick = <T,>(a: readonly T[]) => a[Math.floor(rnd() * a.length)];
const OFF = { kind: 'off' };
const win = (s: number, e: number, b: number) => ({ kind: 'window', start: ((s % 1440) + 1440) % 1440, end: ((e % 1440) + 1440) % 1440, breakMinutes: b });
const q = (n: number) => Math.floor(n / 30) * 30;
// Estruturas com janela com intervalo coberta: no mesmo dia, pela véspera (turno que passa da meia-noite) ou pelo dia
// seguinte (janela 00:00 sem intervalo que cobre o trecho depois da meia-noite).
const coveredWeek = (): any[] => {
  const outer: any[] = [], inner: any[] = [], third: any[] = [];
  const style = pick(['same', 'eve', 'next']);
  for (let d = 0; d < 7; d++) {
    if (rnd() < 0.35) { outer.push(OFF); inner.push(OFF); third.push(OFF); continue; }
    if (style === 'same') {
      const s = q(rnd() * 1440); const L = 30 * int(4, 40);
      outer.push(win(s, s + L, 0));
      const off = 30 * int(0, Math.max(0, L / 30 - 2)); const il = 30 * int(1, Math.max(1, (L - off) / 30));
      inner.push(win(s + off, s + off + il, Math.min(il - 15, 15 * int(1, 8))));
      third.push(rnd() < 0.3 ? win(q(rnd() * 1440), q(rnd() * 1440) + 30, 0) : OFF);
    } else if (style === 'eve') {
      outer.push(win(1080 + 30 * int(0, 10), 30 * int(4, 12), 0));
      inner.push(OFF);
      third.push(OFF);
    } else {
      outer.push(win(1200 + 30 * int(0, 7), 0, 0));
      inner.push(win(1260 + 30 * int(0, 5), 30 * int(1, 8), 15 * int(1, 6)));
      third.push(OFF);
    }
  }
  if (style === 'eve') {
    // A janela com intervalo da manhã seguinte fica dentro do turno da véspera, cruzando outra janela da manhã.
    for (let d = 0; d < 7; d++) {
      const prev = outer[(d + 6) % 7];
      if (prev.kind !== 'window' || rnd() < 0.3) continue;
      const e = prev.end;
      const s = q(rnd() * Math.max(30, e - 30));
      inner[d] = win(s, Math.min(e, s + 30 * int(1, 6)), 15 * int(1, 2));
      third[d] = rnd() < 0.7 ? win(s + 15, e + 30 * int(1, 6), 0) : OFF;
    }
  }
  if (style === 'next') {
    for (let d = 0; d < 7; d++) if (inner[(d + 6) % 7].kind === 'window') third[d] = win(0, 30 * int(2, 16), 0);
  }
  return [{ kind: 'weekly', days: outer }, { kind: 'weekly', days: inner }, { kind: 'weekly', days: third }];
};
const cyclicCover = (): any[] => {
  // Escala cíclica que passa da meia-noite e uma rotina semanal coberta por ela em alguns dias.
  const workDays = int(1, 4), restDays = int(1, 4);
  const shift = win(1200 + 30 * int(0, 6), 30 * int(6, 14), 0);
  const morning = Array.from({ length: 7 }, () => (rnd() < 0.5 ? win(30 * int(0, 4), 30 * int(5, 8), 15 * int(1, 4)) : OFF));
  return [{ kind: 'cyclic', reference: dates.addDays('2026-01-01', int(0, 40)), workDays, restDays, day: shift }, { kind: 'weekly', days: morning }];
};
const TODAY = pick(['2026-11-05', '2027-01-10', '2026-03-31']);
const monthsBetween = (a: string, b: string) => { const out: string[] = []; for (let m: string | null = a; m !== null && m <= b; m = dates.shiftMonthKey(m, 1)) out.push(m); return out; };
let accepted = 0, refused = 0, unsound = 0, monthsChecked = 0, rejectedSamples = 0;
const examples: any[] = [];
const t0 = performance.now();
let maxMs = 0;
for (let s = 0; s < SAMPLES; s++) {
  const today = pick(['2026-11-05', '2027-01-10', '2026-03-31']);
  const firstMonth = dates.shiftMonthKey(today.slice(0, 7), -12);
  const routines = rnd() < 0.7 ? coveredWeek() : cyclicCover();
  if (cal.routinesAmbiguity(routines, today) !== null) { rejectedSamples++; continue; }
  const exceptions: any[] = [];
  const horizonEnd = dates.shiftMonthKey(today.slice(0, 7), 14) as string;
  const randDay = () => dates.addDays(`${firstMonth}-01`, int(-3, 800));
  for (let i = 0; i < int(0, 6); i++) {
    const start = randDay(); const end = dates.addDays(start, pick([0, 0, 0, 1, 2, 6]));
    const kind = pick(['day_off', 'vacation', 'absence', 'reduced', 'schedule_change', 'holiday_worked', 'on_call', 'overtime']);
    if (kind === 'schedule_change' || kind === 'holiday_worked' || kind === 'on_call' || kind === 'overtime') {
      exceptions.push(kind === 'holiday_worked' ? { kind, date: start } : kind === 'schedule_change' ? { kind, date: start, minutes: 30 * int(0, 48) } : { kind, date: start, minutes: 30 * int(1, 48) });
    } else if (kind === 'reduced') exceptions.push({ kind, start, end, minutes: 30 * int(1, 48) });
    else exceptions.push({ kind, start, end });
  }
  const localHolidays: any[] = [];
  if (rnd() < 0.5) localHolidays.push(rnd() < 0.5 ? { kind: 'annual', month: int(1, 12), day: int(1, 28) } : { kind: 'date', date: randDay() });
  if (rnd() < 0.15) localHolidays.push({ kind: 'annual', month: 12, day: pick([30, 31]) });
  const c = { routines, exceptions, localHolidays, worksOnHolidays: rnd() < 0.35 };
  const started = performance.now();
  const verdict = cal.calendarAmbiguity(c, today);
  maxMs = Math.max(maxMs, performance.now() - started);
  if (verdict !== null) { refused++; continue; }
  accepted++;
  // Meses da janela e um horizonte (onde as exceções podem cair), mais meses distantes.
  const months = [...monthsBetween(firstMonth as string, horizonEnd), '2030-02', '2035-07', '2048-12', '2099-11'];
  for (const m of months) {
    for (const periodStart of [false, true]) {
      monthsChecked++;
      const amb = cal.monthPlan(c, m, { periodStart }).ambiguousDays;
      if (amb.length > 0) {
        unsound++;
        if (examples.length < 4) examples.push({ sample: s, today, month: m, periodStart, ambiguousDays: amb, calendar: c });
      }
    }
  }
}
console.log(JSON.stringify({ samples: SAMPLES, routinesAmbiguousSkipped: rejectedSamples, accepted, refused, monthsChecked, unsound, maxConferidorMs: Math.round(maxMs), ms: Math.round(performance.now() - t0) }));
for (const e of examples) console.log(JSON.stringify(e));
