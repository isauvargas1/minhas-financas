// Revisão 2 (P5-F.1-R1), B02: busca adversarial própria da monotonia.
// "Acrescentar uma rotina válida nunca reduz o total da janela, com periodStart no primeiro mês."
// Uso: node --experimental-strip-types --disable-warning=MODULE_TYPELESS_PACKAGE_JSON b02-mono-adv.mts <raiz> [amostras] [semente] [modo]
// modo: mix (padrão) | cluster (dias fora de um bloco em férias) | free (sem férias globais) | spill (motivo de intervalo
// que passa da meia-noite, coberto pelo dia seguinte) | reduced (jornada reduzida vizinha de dias que seguem a rotina)
const root = process.argv[2];
const SAMPLES = Number(process.argv[3] ?? 20000);
let seed = Number(process.argv[4] ?? 7);
const MODE = process.argv[5] ?? 'mix';
const cal = await import(`${root}/src/modules/work-time/calendar.ts`);
const dates = await import(`${root}/src/modules/work-time/dates.ts`);
const rnd = () => { seed = (seed + 0x6d2b79f5) | 0; let v = Math.imul(seed ^ (seed >>> 15), 1 | seed); v = (v + Math.imul(v ^ (v >>> 7), 61 | v)) ^ v; return ((v ^ (v >>> 14)) >>> 0) / 4294967296; };
const int = (a: number, b: number) => a + Math.floor(rnd() * (b - a + 1));
const pick = <T,>(a: readonly T[]) => a[Math.floor(rnd() * a.length)];
const OFF = { kind: 'off' } as const;
const GRIDS = [15, 30, 60, 60, 120];
const win = (start: number, end: number, breakMinutes: number) => ({ kind: 'window', start: ((start % 1440) + 1440) % 1440, end: ((end % 1440) + 1440) % 1440, breakMinutes });
const len = (s: number, e: number) => (e > s ? e - s : 1440 - s + e);
const randBreak = (L: number, g: number) => {
  const r = rnd();
  if (r < 0.45) return 0;
  if (r < 0.7) return Math.min(L - 1, g * int(1, 4));
  return Math.max(0, Math.min(L - 1, g * int(1, Math.max(1, Math.floor((L - 1) / g)))));
};
const anyWindow = () => {
  const g = pick(GRIDS);
  const s = g * int(0, Math.floor(1439 / g));
  const r = rnd();
  const L = r < 0.08 ? 1440 : r < 0.4 ? g * int(1, Math.floor(1440 / g)) : g * int(1, Math.max(1, Math.floor(720 / g)));
  const e = (s + L) % 1440;
  return win(s, e, randBreak(len(s, e), g));
};
// Janela que passa da meia-noite com intervalo maior que o trecho antes dela (o intervalo, na convenção, cai depois da meia-noite).
const spillWindow = () => {
  const s = 1440 - 30 * int(1, 8);
  const after = 30 * int(1, 12);
  const p = 1440 - s;
  const b = Math.min(p + after - 15, p + 15 * int(1, Math.max(1, Math.floor((after - 15) / 15))));
  return win(s, after, Math.max(0, b));
};
const daySpec = (): any => {
  const r = rnd();
  if (r < 0.28) return OFF;
  if (r < 0.38) return { kind: 'duration', minutes: pick([60, 120, 240, 480, 600, 720, 1080, 1320, 1440]) };
  if (r < 0.5) return spillWindow();
  return anyWindow();
};
const workSpec = (): any => { let s = daySpec(); while (s.kind === 'off') s = daySpec(); return s; };
const routineOf = (): any => {
  if (rnd() < 0.72) return { kind: 'weekly', days: Array.from({ length: 7 }, daySpec) };
  return { kind: 'cyclic', reference: dates.addDays('2026-07-20', int(0, 30)), workDays: int(1, 4), restDays: int(1, 3), day: workSpec() };
};
// Rotina acrescentada: às vezes um só dia da semana, às vezes uma cópia de outra com outro intervalo, às vezes uma janela
// que cobre a meia-noite entre dois dias (00:00 do dia seguinte, sem intervalo).
const addedRoutineOf = (base: any[]): any => {
  const r = rnd();
  if (r < 0.25) { const d = int(0, 6); return { kind: 'weekly', days: Array.from({ length: 7 }, (_, i) => (i === d ? workSpec() : OFF)) }; }
  if (r < 0.4 && base.length > 0) {
    const src = pick(base);
    if (src.kind === 'weekly') return { kind: 'weekly', days: src.days.map((d: any) => (d.kind === 'window' ? { ...d, breakMinutes: rnd() < 0.5 ? 0 : randBreak(len(d.start, d.end), 15) } : d)) };
  }
  if (r < 0.55) { const d = int(0, 6); return { kind: 'weekly', days: Array.from({ length: 7 }, (_, i) => (i === d ? win(0, 30 * int(1, 16), 0) : OFF)) }; }
  return routineOf();
};
const MONTHS_SETS = [['2026-10'], ['2026-09', '2026-10'], ['2026-08', '2026-09', '2026-10'], ['2026-02', '2026-03'], ['2026-11']];
const exceptionsFor = (months: string[], mode: string): any[] => {
  const first = `${months[0]}-01`;
  const lastMonth = months[months.length - 1];
  const daysTotal = months.reduce((n, m) => n + dates.datesOfMonth(m).length, 0);
  const out: any[] = [];
  const minutes = () => pick([60, 120, 240, 360, 480, 600, 720, 900, 1080, 1200, 1380, 1440, 30 * int(1, 48)]);
  // Bloco de dias em que as exceções se concentram (no começo do período, metade das vezes).
  const clusterStart = rnd() < 0.5 ? first : dates.addDays(first, int(0, daysTotal - 1));
  const clusterLen = int(2, 6);
  const cluster = Array.from({ length: clusterLen }, (_, i) => dates.addDays(clusterStart, i));
  const usedSingle = new Map<string, Set<string>>();
  const once = (kind: string, date: string) => { const s = usedSingle.get(kind) ?? new Set(); if (s.has(date)) return false; s.add(date); usedSingle.set(kind, s); return true; };
  const rangeTaken = new Map<string, Set<string>>();
  const rangeFree = (kind: string, ds: string[]) => { const s = rangeTaken.get(kind) ?? new Set(); if (ds.some((d) => s.has(d))) return false; ds.forEach((d) => s.add(d)); rangeTaken.set(kind, s); return true; };
  if (mode === 'cluster' || (mode === 'mix' && rnd() < 0.35)) {
    // Férias em todo o período, fora do bloco.
    const last = dates.addDays(`${lastMonth}-01`, dates.datesOfMonth(lastMonth).length - 1);
    let start: string | null = null;
    for (let d = dates.addDays(first, -3); d <= dates.addDays(last, 2); d = dates.addDays(d, 1)) {
      const inCluster = cluster.includes(d);
      if (!inCluster && start === null) start = d;
      if (inCluster && start !== null) { out.push({ kind: 'vacation', start, end: dates.addDays(d, -1) }); rangeFree('vacation', []); start = null; }
    }
    if (start !== null) out.push({ kind: 'vacation', start, end: dates.addDays(last, 2) });
  }
  const pickDay = () => (rnd() < 0.8 ? pick(cluster) : dates.addDays(first, int(-2, daysTotal + 1)));
  const n = int(1, mode === 'reduced' ? 10 : 8);
  for (let i = 0; i < n; i++) {
    const d = pickDay();
    const span = rnd() < 0.7 ? 1 : int(2, 3);
    const ds = Array.from({ length: span }, (_, k) => dates.addDays(d, k));
    const end = ds[ds.length - 1];
    const r = rnd();
    const pReduced = mode === 'reduced' ? 0.45 : 0.22;
    if (r < pReduced) { if (rangeFree('reduced', ds)) out.push({ kind: 'reduced', start: d, end, minutes: minutes() }); }
    else if (r < pReduced + 0.18) { if (once('on_call', d)) out.push({ kind: 'on_call', date: d, minutes: minutes() }); }
    else if (r < pReduced + 0.34) { if (once('overtime', d)) out.push({ kind: 'overtime', date: d, minutes: minutes() }); }
    else if (r < pReduced + 0.44) { if (once('schedule_change', d)) out.push({ kind: 'schedule_change', date: d, minutes: rnd() < 0.2 ? 0 : minutes() }); }
    else if (r < pReduced + 0.54) { if (rangeFree('day_off', ds)) out.push({ kind: pick(['day_off', 'absence']), start: d, end }); }
    else if (r < pReduced + 0.62) { if (once('holiday_worked', d)) out.push({ kind: 'holiday_worked', date: d }); }
  }
  return out;
};
const totalOf = (c: any, months: string[]) => {
  let t = 0; let amb = false; const perDay: Record<string, number> = {};
  months.forEach((m, i) => {
    const p = cal.monthPlan(c, m, { periodStart: i === 0 });
    t += p.minutes;
    if ((p.ambiguousDays ?? []).length) amb = true;
    for (const d of p.days) perDay[d.date] = d.minutes;
  });
  return { t, amb, perDay };
};
let checked = 0, viol = 0, ambiguous = 0, reducedCapped = 0, spillPairs = 0;
const examples: any[] = [];
const t0 = performance.now();
for (let s = 0; s < SAMPLES; s++) {
  const mode = MODE === 'mix' ? pick(['cluster', 'free', 'spill', 'reduced']) : MODE;
  const base: any[] = Array.from({ length: int(1, 2) }, routineOf);
  if (mode === 'spill') {
    // Véspera com janela que passa da meia-noite e intervalo depois dela; dia seguinte com janela 00:00 sem intervalo.
    const d = int(0, 6);
    const days = Array.from({ length: 7 }, () => OFF as any);
    days[d] = spillWindow();
    days[(d + 1) % 7] = win(0, 30 * int(2, 16), 0);
    if (rnd() < 0.6) days[d] = win(1440 - 30 * int(1, 8), 0, 0);
    base.push({ kind: 'weekly', days });
  }
  const added = mode === 'spill' && rnd() < 0.6
    ? (() => { const d = int(0, 6); const days = Array.from({ length: 7 }, () => OFF as any); days[d] = spillWindow(); return { kind: 'weekly', days }; })()
    : addedRoutineOf(base);
  if (base.length + 1 > 3 && rnd() < 0.5) base.pop();
  const months = pick(MONTHS_SETS);
  const extra = {
    exceptions: exceptionsFor(months, mode),
    worksOnHolidays: rnd() < 0.4,
    localHolidays: rnd() < 0.5 ? [{ kind: 'date', date: dates.addDays(`${months[0]}-01`, int(-1, 40)) }] : [],
  };
  const before = totalOf({ ...extra, routines: base }, months);
  if (before.amb) { ambiguous++; continue; }
  let addedNow = added;
  let after = totalOf({ ...extra, routines: [...base, addedNow] }, months);
  for (let tries = 0; after.amb && tries < 6; tries++) { addedNow = addedRoutineOf(base); after = totalOf({ ...extra, routines: [...base, addedNow] }, months); }
  if (after.amb) { ambiguous++; continue; }
  checked++;
  if (mode === 'spill') spillPairs++;
  for (const m of months) for (const d of cal.monthPlan({ ...extra, routines: [...base, addedNow] }, m).days) if (d.capped && d.applied.includes('reduced')) reducedCapped++;
  if (after.t < before.t) {
    viol++;
    if (examples.length < 4) {
      const days = Object.keys(before.perDay).filter((k) => before.perDay[k] !== after.perDay[k]).map((k) => `${k}:${before.perDay[k]}→${after.perDay[k]}`);
      examples.push({ sample: s, mode, before: before.t, after: after.t, days, base, added: addedNow, extra, months });
    }
  }
}
console.log(JSON.stringify({ root: root.split('/').slice(-1)[0], samples: SAMPLES, seed: Number(process.argv[4] ?? 7), mode: MODE, checked, ambiguous, spillPairs, reducedCappedDays: reducedCapped, violations: viol, ms: Math.round(performance.now() - t0) }));
for (const e of examples) console.log(JSON.stringify(e));
