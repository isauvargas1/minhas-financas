#!/usr/bin/env python3
"""Revisão 4: alinha, linha a linha, as capturas de página inteira "antes" e "depois" (RGBA cru, gerado pelo
ImageMagick) e lista o que difere fora dos blocos inseridos. Uso: alinhar.py <dir-raw> <nome> [<nome> ...]"""
import difflib, hashlib, json, sys
from pathlib import Path

raw = Path(sys.argv[1])


def load(side, name):
    w, h = map(int, (raw / side / f'{name}.size').read_text().split('x'))
    data = (raw / side / f'{name}.rgba').read_bytes()
    assert len(data) == w * h * 4, (name, side, len(data), w, h)
    rows = [data[y * w * 4:(y + 1) * w * 4] for y in range(h)]
    return w, h, rows


def pixdiff(ra, rb, w):
    """Pixels diferentes entre duas linhas de mesma largura e as colunas extremas."""
    n, xs = 0, []
    for x in range(w):
        if ra[x * 4:x * 4 + 4] != rb[x * 4:x * 4 + 4]:
            n += 1
            xs.append(x)
    return n, (min(xs), max(xs)) if xs else None


out = {}
for name in sys.argv[2:]:
    wa, ha, a = load('antes', name)
    wb, hb, b = load('depois', name)
    assert wa == wb
    ka = [hashlib.blake2b(r, digest_size=12).digest() for r in a]
    kb = [hashlib.blake2b(r, digest_size=12).digest() for r in b]
    sm = difflib.SequenceMatcher(a=ka, b=kb, autojunk=False)
    ops = []
    for tag, i1, i2, j1, j2 in sm.get_opcodes():
        if tag == 'equal':
            continue
        entry = {'op': tag, 'antes_y': [i1, i2], 'depois_y': [j1, j2], 'altura_antes': i2 - i1, 'altura_depois': j2 - j1}
        if tag == 'replace' and i2 - i1 == j2 - j1:
            total, x0, x1, y0, y1 = 0, None, None, None, None
            for k in range(i2 - i1):
                n, span = pixdiff(a[i1 + k], b[j1 + k], wa)
                if n:
                    total += n
                    x0 = span[0] if x0 is None else min(x0, span[0])
                    x1 = span[1] if x1 is None else max(x1, span[1])
                    y0 = k if y0 is None else y0
                    y1 = k
            entry['pixels_diferentes'] = total
            entry['caixa_x'] = [x0, x1]
        ops.append(entry)
    equal_rows = sum(i2 - i1 for tag, i1, i2, j1, j2 in sm.get_opcodes() if tag == 'equal')
    out[name] = {'antes': f'{wa}x{ha}', 'depois': f'{wb}x{hb}', 'linhas_iguais': equal_rows, 'blocos': ops}
print(json.dumps(out, ensure_ascii=False, indent=1))
