// Busca da revisão (b02-busca.mts) com o conferidor do calendário inteiro (`calendarAmbiguity`): configuração
// aceita (conferidor nulo) com dia ambíguo em alguma janela da estimativa? Também confere o dia apontado.
// Cópia da R1 (aceita-causas): conta as recusas cujo dia apontado não é ambíguo no plano do mês, por causa,
// e confere que, com a causa `routines`, as rotinas sozinhas são ambíguas (routinesAmbiguity não nulo).
const root = process.argv[2];
const SAMPLES = Number(process.argv[3] ?? 5000);
let seed = Number(process.argv[4] ?? 7);
const cal = await import(`${root}/src/modules/work-time/calendar.ts`);
const dates = await import(`${root}/src/modules/work-time/dates.ts`);
const rnd = () => { seed = (seed + 0x6d2b79f5) | 0; let v = Math.imul(seed ^ (seed >>> 15), 1 | seed); v = (v + Math.imul(v ^ (v >>> 7), 61 | v)) ^ v; return ((v ^ (v >>> 14)) >>> 0) / 4294967296; };
const pick = <T,>(a: readonly T[]) => a[Math.floor(rnd() * a.length)];
const OFF = { kind: 'off' };
const q = (n: number) => Math.floor(n / 15) * 15;
const baseWin = () => { const s = q(rnd() * 1440); const len = 15 * (8 + Math.floor(rnd() * 88)); return { kind: 'window', start: s, end: (s + len) % 1440, breakMinutes: 0, len }; };
const inside = (b: any) => { const off = q(rnd() * Math.max(15, b.len - 30)); const len = Math.max(30, q(rnd() * (b.len - off))); const s = (b.start + off) % 1440; const br = q(rnd() * (len - 15)); return { kind: 'window', start: s, end: (s + len) % 1440, breakMinutes: Math.max(0, br) }; };
const anyWin = () => { const s = q(rnd() * 1440); const L = 15 * (1 + Math.floor(rnd() * 96)); const b = rnd() < 0.5 ? 0 : q(rnd() * (L - 15)); return { kind: 'window', start: s, end: (s + L) % 1440, breakMinutes: b }; };
const strip = (w: any) => ({ kind: 'window', start: w.start, end: w.end, breakMinutes: w.breakMinutes });
const weeklyOf = (specs: Record<number, any>) => ({ kind: 'weekly', days: Array.from({ length: 7 }, (_, d) => specs[d] ?? OFF) });
const genConfig = () => {
  const routines: any[] = []; const bases: Record<number, any> = {}; const ins: Record<number, any> = {};
  for (let d = 0; d < 7; d++) if (rnd() < 0.6) { const b = baseWin(); bases[d] = strip(b); if (rnd() < 0.7) ins[d] = inside(b); }
  routines.push(weeklyOf(bases)); if (Object.keys(ins).length) routines.push(weeklyOf(ins));
  if (rnd() < 0.5) routines.push(rnd() < 0.5 ? weeklyOf(Object.fromEntries(Array.from({ length: 7 }, (_, d) => [d, rnd() < 0.5 ? anyWin() : OFF]))) : { kind: 'cyclic', reference: dates.addDays('2026-01-01', Math.floor(rnd() * 20)), workDays: 1 + Math.floor(rnd() * 3), restDays: Math.floor(rnd() * 3), day: anyWin() });
  return routines;
};
const randDate = () => dates.addDays('2025-10-20', Math.floor(rnd() * 500));
const randExceptions = () => {
  const out: any[] = []; const used: Record<string, [string, string][]> = {};
  for (let i = 0; i < Math.floor(rnd() * 8); i++) {
    const start = randDate(); const end = dates.addDays(start, Math.floor(rnd() * 3)); const minutes = 15 * (1 + Math.floor(rnd() * 96));
    const kind = pick(['vacation', 'day_off', 'absence', 'holiday_worked', 'on_call', 'overtime', 'reduced', 'schedule_change'] as const);
    const span: [string, string] = ['vacation', 'day_off', 'absence', 'reduced'].includes(kind) ? [start, end] : [start, start];
    if ((used[kind] ?? []).some(([a, b]) => span[0] <= b && a <= span[1])) continue;
    (used[kind] ??= []).push(span);
    switch (kind) {
      case 'vacation': case 'day_off': case 'absence': out.push({ kind, start, end }); break;
      case 'holiday_worked': out.push({ kind, date: start }); break;
      case 'on_call': case 'overtime': out.push({ kind, date: start, minutes }); break;
      case 'reduced': out.push({ kind, start, end, minutes }); break;
      default: out.push({ kind, date: start, minutes: Math.min(1440, minutes) });
    }
  }
  return out;
};
const TODAY = '2026-11-05';
const months: string[] = []; for (let m = '2025-11'; m <= '2027-06'; m = dates.shiftMonthKey(m, 1)!) months.push(m);
const notConfirmedBy: Record<string, number> = {}; const suspendedBy: Record<string, number> = {};
let checkerMs = 0, worstMs = 0; let accepted = 0, acceptedButAmbiguous = 0, refused = 0, refusedConfirmed = 0, t0 = performance.now();
const examples: any[] = [];
for (let s = 0; s < SAMPLES; s++) {
  const calendar = { routines: genConfig(), exceptions: randExceptions(), worksOnHolidays: rnd() < 0.4, localHolidays: rnd() < 0.4 ? [{ kind: 'date', date: randDate() }, { kind: 'annual', month: 1 + Math.floor(rnd() * 12), day: 1 + Math.floor(rnd() * 28) }] : [] };
  const tc = performance.now(); const verdict = cal.calendarAmbiguity(calendar, TODAY); const spent = performance.now() - tc; checkerMs += spent; if (spent > worstMs) worstMs = spent;
  // Todo dia de 2025-11 a 2027-06, como mês avulso e como abertura do período.
  let ambiguousSomewhere: string | null = null;
  for (const m of months) {
    for (const periodStart of [false, true]) {
      const plan = cal.monthPlan(calendar, m, { periodStart });
      if (plan.ambiguousDays.length > 0 && ambiguousSomewhere === null) ambiguousSomewhere = `${plan.ambiguousDays[0]}${periodStart ? ' (abre o período)' : ''}`;
    }
  }
  if (verdict === null) { accepted++; if (ambiguousSomewhere) { acceptedButAmbiguous++; if (examples.length < 3) examples.push({ ambiguousSomewhere, calendar }); } }
  else {
    refused++;
    // O dia apontado é mesmo ambíguo (como mês avulso, ou abrindo o período quando a causa é essa)?
    const plan = cal.monthPlan(calendar, verdict.date.slice(0, 7), { periodStart: verdict.cause.kind === 'period_start' });
    if (plan.ambiguousDays.includes(verdict.date)) refusedConfirmed++;
    else {
      const key = verdict.cause.kind + (verdict.cause.kind === 'routines' ? (cal.routinesAmbiguity(calendar.routines, TODAY) !== null ? ':rotinas-sozinhas-ambiguas' : ':ROTINAS-SOZINHAS-SEGURAS') : ':DIA-NAO-AMBIGUO');
      notConfirmedBy[key] = (notConfirmedBy[key] ?? 0) + 1;
      const day = cal.dayPlan(calendar, verdict.date);
      const why = `${day.outcome}${day.holiday ? ':feriado' : ''}${day.applied?.length ? ':' + day.applied.join('+') : ''}`;
      suspendedBy[why] = (suspendedBy[why] ?? 0) + 1;
    }
  }
}
console.log(JSON.stringify({ samples: SAMPLES, accepted, acceptedButAmbiguous, refused, refusedConfirmed, ms: Math.round(performance.now() - t0), conferidorMedioMs: +(checkerMs / SAMPLES).toFixed(2), conferidorPiorMs: +worstMs.toFixed(1), notConfirmedBy, suspendedBy }));
for (const e of examples) console.log(JSON.stringify(e).slice(0, 1500));
