// Variante dirigida (revisão da R1, N01): a rotina acrescentada só tem janela num dia da semana D, e as outras
// ocorrências de D no período estão em férias; nas que sobram, jornada reduzida com plantão ou hora extra.
const root = process.argv[2];
const SAMPLES = Number(process.argv[3] ?? 1500);
const cal = await import(`${root}/src/modules/work-time/calendar.ts`);
const dates = await import(`${root}/src/modules/work-time/dates.ts`);
let seed = Number(process.argv[4] ?? 20261110);
const next = () => { seed = (seed + 0x6d2b79f5) | 0; let v = Math.imul(seed ^ (seed >>> 15), 1 | seed); v = (v + Math.imul(v ^ (v >>> 7), 61 | v)) ^ v; return ((v ^ (v >>> 14)) >>> 0) / 4294967296; };
const OFF = { kind: 'off' };
const PERIOD = ['2026-08', '2026-09', '2026-10'];
const plans = (c: any) => PERIOD.map((m, i) => cal.monthPlan(c, m, { periodStart: i === 0 }));
const total = (ps: any[]) => ps.reduce((s, p) => s + p.minutes, 0);
const hours = (min: number, max: number) => (min + Math.floor(next() * (max - min + 1))) * 60;
const sameDay = (min: number, max: number) => { const len = hours(min, max); const start = Math.floor(next() * ((1440 - len) / 60 + 1)) * 60; return { kind: 'window', start, end: (start + len) % 1440, breakMinutes: 0 }; };
let checked = 0, violations = 0;
for (let s = 0; s < SAMPLES; s++) {
  const D = Math.floor(next() * 7);
  const base = { kind: 'weekly', days: Array.from({ length: 7 }, () => (next() < 0.8 ? sameDay(2, 10) : OFF)) };
  const added = { kind: 'weekly', days: Array.from({ length: 7 }, (_, d) => (d === D ? sameDay(1, 8) : OFF)) };
  const all: string[] = [];
  for (let date = '2026-08-01'; date <= '2026-10-31'; date = dates.addDays(date, 1)) if (dates.weekdayIndex(date) === D) all.push(date);
  const keep = new Set(all.filter(() => next() < 0.25));
  const exceptions: any[] = [];
  for (const date of all) {
    if (!keep.has(date)) { exceptions.push({ kind: 'vacation', start: date, end: date }); continue; }
    exceptions.push({ kind: 'reduced', start: date, end: date, minutes: hours(4, 16) });
    if (next() < 0.5) exceptions.push({ kind: 'on_call', date, minutes: 1440 });
    else { exceptions.push({ kind: 'on_call', date, minutes: hours(6, 24) }); exceptions.push({ kind: 'overtime', date, minutes: hours(6, 24) }); }
  }
  const extra = { exceptions, worksOnHolidays: true, localHolidays: [] };
  const b = plans({ ...extra, routines: [base] }); const a = plans({ ...extra, routines: [base, added] });
  if ([...b, ...a].some((p) => p.ambiguousDays.length > 0)) continue;
  checked++;
  if (total(a) < total(b)) violations++;
}
console.log(JSON.stringify({ samples: SAMPLES, checked, violations }));
