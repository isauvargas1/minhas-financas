def prepare(name: str) -> str:
    target = f'{OUT}/{name}'
    shutil.rmtree(target, ignore_errors=True)
    os.makedirs(f'{target}/tests/unit')
    os.makedirs(f'{target}/functions')
    shutil.copytree(f'{WT}/src', f'{target}/src')
    shutil.copytree(f'{WT}/functions/src', f'{target}/functions/src')
    for path in ('firestore.rules', 'package.json'):
        shutil.copy2(f'{WT}/{path}', f'{target}/{path}')
    for path in (CALC, PREF, GUARD):
        shutil.copy2(f'{WT}/{path}', f'{target}/{path}')
    os.symlink(f'{WT}/node_modules', f'{target}/node_modules')
    return target


def apply(target: str, path: str, old, new: str) -> None:
    file = f'{target}/{path}'
    with open(file, encoding='utf-8') as handle:
        source = handle.read()
    if old is APPEND:
        source = source + ('' if source.endswith('\n') else '\n') + new
    elif old is None:
        source = new + source
    else:
        count = source.count(old)
        if count != 1:
            sys.exit(f'mutante inválido: {path} contém o trecho {count} vez(es): {old[:80]!r}')
        source = source.replace(old, new)
    with open(file, 'w', encoding='utf-8') as handle:
        handle.write(source)


def run(target: str, test_file: str) -> tuple[int, list[str]]:
    result = subprocess.run(
        ['node', '--experimental-strip-types', '--disable-warning=MODULE_TYPELESS_PACKAGE_JSON', '--test', test_file],
        cwd=target, capture_output=True, text=True, timeout=300)
    failed = [line[len('not ok '):] for line in result.stdout.splitlines() if line.startswith('not ok ')]
    return result.returncode, failed


def main() -> None:
    os.makedirs(OUT, exist_ok=True)
    rows = []
    pristine = prepare('original')
    for test_file in (CALC, PREF, GUARD):
        code, failed = run(pristine, test_file)
        print(f'ORIGINAL {test_file}: saída {code}, falhas {failed}')
        if code != 0:
            sys.exit('a cópia original precisa passar')
    shutil.rmtree(pristine)
    for mutant_id, description, test_file, edits in MUTANTS:
        target = prepare(mutant_id)
        for path, old, new in edits:
            apply(target, path, old, new)
        code, failed = run(target, test_file)
        killed = code != 0
        rows.append({'id': mutant_id, 'mutante': description, 'teste': test_file, 'morto': killed,
                     'saida': code, 'falhas': failed})
        print(f"{mutant_id:<4} {'MORTO ' if killed else 'VIVO  '} {description} -> {failed[:2]}")
        shutil.rmtree(target)
    with open(f'{OUT}/resultado.json', 'w', encoding='utf-8') as handle:
        json.dump(rows, handle, ensure_ascii=False, indent=2)
    dead = sum(1 for row in rows if row['morto'])
    print(f'mortos {dead}/{len(rows)}')


if __name__ == '__main__':
    main()
