// Sonda do revisor (R2A, item 2): C e C' = C + rotina, ambos com cálculo seguro, nunca com menos horas em C'
// na janela da estimativa (o primeiro mês abrindo o período).
// Uso: node --experimental-strip-types s01-monotonia.mts <raiz> <amostras> <semente> <perfil> [fino 0|1]
import { rng, genRoutine, genExceptions } from './lib.mts';

const [root, samplesArg, seedArg, profileArg, fineArg] = process.argv.slice(2);
const cal = await import(`${root}/src/modules/work-time/calendar.ts`);
const dates = await import(`${root}/src/modules/work-time/dates.ts`);
const N = Number(samplesArg ?? 2000);
const next = rng(Number(seedArg ?? 1));
const profile = (profileArg ?? 'mix') as any;
const fine = fineArg === '1';

// Janelas: (hoje, meses completos antes do mês de hoje) e o mês corrente.
const windowsFor = (today: string): string[][] => {
  const current = dates.monthKeyOf(today);
  const out: string[][] = [[current]];
  for (const n of [1, 3, 6, 12]) {
    const months: string[] = [];
    for (let k = n; k >= 1; k -= 1) months.push(dates.shiftMonthKey(current, -k));
    out.push(months);
  }
  return out;
};
const TODAYS = ['2026-11-06', '2026-12-15', '2027-01-10'];

const stats = { samples: 0, pairsSafeWindow: 0, pairsAccepted: 0, violationsSafeWindow: 0, violationsAccepted: 0, withReducedNight: 0, withUncertainEve: 0, transitions: 0 };
const uncertainEves = (c: any, months: string[]) => { const out = new Set<string>(); for (const e of c.exceptions) if (e.kind === 'reduced') for (let d = e.start; d <= e.end; d = dates.addDays(d, 1)) { if (!months.includes(d.slice(0, 7))) continue; const nb = cal._neighborWindowsOn(c, d); if (!nb.certain && nb.windows.some((w: any) => w.end > (dates.dayNumber(d) + 1) * 1440)) out.add(d); } return out; };
const examples: any[] = [];
const t0 = Date.now();
for (let sample = 0; sample < N; sample += 1) {
  stats.samples += 1;
  const base = Array.from({ length: 1 + Math.floor(next() * 2) }, () => genRoutine(next, profile, fine));
  const added = genRoutine(next, profile, fine);
  const exceptions = genExceptions(next, dates, '2026-09-25', 100);
  const localHolidays = next() < 0.2 ? [{ kind: 'date', date: dates.addDays('2026-09-25', Math.floor(next() * 100)) }] : [];
  const extra = { exceptions, localHolidays, worksOnHolidays: next() < 0.5 };
  const C = { ...extra, routines: base };
  const C2 = { ...extra, routines: [...base, added] };
  for (const today of TODAYS) {
    const acceptedBoth = cal.calendarAmbiguity(C, today) === null && cal.calendarAmbiguity(C2, today) === null;
    for (const months of windowsFor(today)) {
      const plans = (c: any) => months.map((m, i) => cal.monthPlan(c, m, { periodStart: i === 0 }));
      const p1 = plans(C);
      const p2 = plans(C2);
      if ([...p1, ...p2].some((p) => p.ambiguousDays.length > 0)) continue;
      const h1 = p1.reduce((s, p) => s + p.minutes, 0);
      const h2 = p2.reduce((s, p) => s + p.minutes, 0);
      stats.pairsSafeWindow += 1;
      if (acceptedBoth) stats.pairsAccepted += 1;
      if (p2.some((p) => p.days.some((d: any) => d.applied.includes('reduced')))) stats.withReducedNight += 1;
      const u1 = uncertainEves(C, months); const u2 = uncertainEves(C2, months);
      if (u1.size + u2.size > 0) stats.withUncertainEve += 1;
      if ([...u1].some((d) => !u2.has(d)) || [...u2].some((d) => !u1.has(d))) stats.transitions += 1;
      if (h2 < h1) {
        stats.violationsSafeWindow += 1;
        if (acceptedBoth) stats.violationsAccepted += 1;
        // dia a dia, onde perdeu
        const diffs: any[] = [];
        p1.forEach((p, i) => p.days.forEach((d: any, j: number) => { const e = p2[i].days[j]; if (e.minutes !== d.minutes) diffs.push([d.date, d.minutes, e.minutes]); }));
        if (examples.length < 8) examples.push({ today, months, h1, h2, acceptedBoth, diffs: diffs.slice(0, 12), C: JSON.stringify(C), added: JSON.stringify(added) });
      }
    }
  }
}
console.log(JSON.stringify({ profile, fine, seed: Number(seedArg ?? 1), ms: Date.now() - t0, stats }, null, 1));
for (const e of examples) console.log(JSON.stringify(e));
